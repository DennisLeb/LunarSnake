cmake_minimum_required(VERSION 3.16)

set(CMAKE_CXX_STANDARD 20)


# This creates an option inside of 'CMakeSettings.json' or when building the project in the console
# Basically this will change the absolute ASSET_PATH from your machine's files to be the local path
option(PRODUCTION_BUILD "Make this a production build" OFF)
# Delete the 'out' folder after changing, if VS doesn't recognize the change!

if(MSVC) 

	set(CMAKE_MSVC_RUNTIME_LIBRARY "MultiThreaded$<$<CONFIG:Debug>:Debug>")
	add_compile_options(/arch:AVX2) # Make sure SIMD optimizations take place

endif()

if (PRODUCTION_BUILD)

	set(CMAKE_INTERPROCEDURAL_OPTIMIZATION TRUE) # # Link time optimization for release/production builds

endif()

# Ensure SFML is built as static libraries (primarily viable for small projects)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "Build Raylib as static libraries" FORCE)


project("LunarSnake")


# Handle raylib dependency
include("dependencies.cmake")


# 'MY_SOURCES' is defined to be a list of all the source files for the project
# DON'T ADD THE SOURCES BY HAND IF ASKED, they are already added with this macro
file(GLOB_RECURSE MY_SOURCES CONFIGURE_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/src/*.cpp")


add_executable(${CMAKE_PROJECT_NAME} ${MY_SOURCES})

set_property(TARGET ${CMAKE_PROJECT_NAME} PROPERTY CXX_STANDARD 20)



if (PRODUCTION_BUILD)

	# Set up the 'ASSET_PATH' macro to be in the root folder of your .exe
	target_compile_definitions("${CMAKE_PROJECT_NAME}" PUBLIC ASSET_PATH="./assets/") 

	# Remove the option to debug asserts (?)
	target_compile_definitions("${CMAKE_PROJECT_NAME}" PUBLIC PRODUCTION_BUILD=1) 

else()

	# This is useful to get an 'ASSETS_PATH' in your IDE during development
	target_compile_definitions("${CMAKE_PROJECT_NAME}" PUBLIC ASSET_PATH="${CMAKE_CURRENT_SOURCE_DIR}/assets/")
	target_compile_definitions("${CMAKE_PROJECT_NAME}" PUBLIC PRODUCTION_BUILD=0) 

endif()


if(MSVC)

	# This line disables compiler warnings
	#target_compile_definitions("${CMAKE_PROJECT_NAME}" PUBLIC _CRT_SECURE_NO_WARNINGS)

	if (PRODUCTION_BUILD)

		# YOU CAN REMOVE THE CONSOLE WITH THIS LINE!
		set_target_properties("${CMAKE_PROJECT_NAME}" PROPERTIES LINK_FLAGS "/SUBSYSTEM:WINDOWS /ENTRY:mainCRTStartup") #no console
	
	endif()

endif()

# Change the icon of the executable file
if (WIN32)

    target_sources(${CMAKE_PROJECT_NAME}
        PRIVATE
        "assets/executable/app.rc"
    )

endif()


target_include_directories("${CMAKE_PROJECT_NAME}" PUBLIC "${CMAKE_CURRENT_SOURCE_DIR}/src")


# Link raylib library. Here you would add other libraries!
target_link_libraries("${CMAKE_PROJECT_NAME}" PRIVATE raylib)
